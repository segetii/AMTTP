\section{Computational Analysis of the Adaptive Viscosity
  Mechanism}\label{app:computational_ns}

The regularity theorems in Section~\ref{sec:ns} are conditional:
Theorem~\ref{thm:ns_regularity} assumes an \emph{a priori}
enstrophy bound, and Theorem~\ref{thm:ns_strong} requires
small data.  This appendix reports quantitative simulation
results that (i) identify the \emph{mechanism} by which adaptive
viscosity achieves stability, (ii) delineate the precise parameter
regime where a quadratic damping inequality holds, and (iii)
formulate testable conjectures toward unconditional results.

All simulations use the Taylor--Green initial condition
$\bm{u}_0 = (\sin x\cos y\cos z,\,-\cos x\sin y\cos z,\,0)$
on $\T^3$, pseudo-spectral discretisation at $N=128$ with
$2/3$-rule de-aliasing, semi-implicit integrator, $\theta=1$,
and $\mathrm{dt} = 5\times 10^{-4}$.

%% ──────────────────────────────────────────────────────────────────
\subsection{Notation and the Implemented System}
\label{app:ns:notation}
%% ──────────────────────────────────────────────────────────────────

The adaptive-viscosity system solved in the simulations is
\begin{equation}\label{eq:app_ns}
  \partial_t\bm{u}+(\bm{u}\cdot\grad)\bm{u}
  = -\grad p + \nu_{\mathrm{eff}}(t)\,\Delta\bm{u},\quad
  \nu_{\mathrm{eff}}(t) = \nu_0\bigl(1+\gamma^*(E_{\mathrm{BS}}(t))\bigr),\quad
  \gamma^*(E) = \frac{E}{E+\theta},
\end{equation}
where $\theta > 0$ is a sensitivity parameter.  The constant-viscosity
case sets $\gamma^* \equiv 0$, giving $\nu_{\mathrm{eff}} = \nu_0$.

\paragraph{Implemented BSDT channels.}
The simulation computes three of the four channels from
Section~\ref{sec:ns:channels}:
\begin{align}
  \delta_C(t) &= \frac{|\mathcal{E}(t) - \bar{\mathcal{E}}_t|}
  {\max(\sigma_{\mathcal{E},t},\,10^{-10})},
  &\text{(enstrophy z-score)}\label{eq:app_dC}\\
  \delta_G(t) &= \Bigl[\frac{1}{|\mathcal{K}|}
  \sum_{k\in\mathcal{K}}\bigl(\log E(k,t)
  - (-\tfrac{5}{3}\log k + b_t)\bigr)^2\Bigr]^{1/2},
  &\text{(spectral departure from $k^{-5/3}$)}\label{eq:app_dG}\\
  \delta_T(t) &= \frac{\norm{\hat{\bm{u}}(t)-\hat{\bm{u}}(t-\Delta t)}_{L^2}}
  {\norm{\hat{\bm{u}}(t)}_{L^2} + 10^{-20}},
  &\text{(relative spectral change rate)}\label{eq:app_dT}
\end{align}
where $\bar{\mathcal{E}}_t$ and $\sigma_{\mathcal{E},t}$ are running
mean and standard deviation of enstrophy over all previous snapshots.
The vorticity-strain alignment channel $\delta_A$ is \textbf{not
computed} by the solver ($\delta_A \equiv 0$).  The blind-spot energy
is
\begin{equation}\label{eq:app_ebs}
  E_{\mathrm{BS}}(t) = \delta_C(t)^2 + \delta_G(t)^2 + \delta_T(t)^2.
\end{equation}
This differs from the paper's definition~\eqref{eq:ebs_ns} in three
respects: (a) $\delta_C$ is a z-score rather than raw enstrophy,
(b) the weights $w_i$ are uniform (not Fisher-optimised), and
(c) the pairwise interaction term $L_{\mathrm{pair}}$ is absent.
These differences should be borne in mind when comparing simulation
results to the theoretical framework of Section~\ref{sec:ns}.

%% ──────────────────────────────────────────────────────────────────
\subsection{Viscosity Bounds and Leray Well-Posedness}
\label{app:ns:bounds}
%% ──────────────────────────────────────────────────────────────────

\begin{proposition}[Viscosity Bounds]\label{prop:nu_bounds}
For any $\theta > 0$ and $E_{\mathrm{BS}} \ge 0$:
\[
  \nu_0 \;\le\; \nu_{\mathrm{eff}}(t) \;<\; 2\nu_0
  \qquad\text{for all } t \ge 0.
\]
The adaptive mechanism provides at most a factor-$2$ increase
in viscosity.
\end{proposition}

\begin{proof}
$\gamma^*(E) = E/(E+\theta)$ with $E \ge 0$ and $\theta > 0$.
The minimum $\gamma^*(0) = 0$ gives $\nu_{\mathrm{eff}} = \nu_0$.
As $E \to \infty$, $\gamma^* \to 1^-$, so
$\nu_{\mathrm{eff}} \to 2\nu_0^-$.
\end{proof}

\begin{corollary}[Leray Theory Applies]
Since $\nu_{\mathrm{eff}} \ge \nu_0 > 0$ is bounded and measurable,
Leray--Hopf weak solutions exist globally.  Uniqueness and smoothness
remain open (as for classical NS).
\end{corollary}

%% ──────────────────────────────────────────────────────────────────
\subsection{Rapid Saturation of Adaptive Control}
\label{app:ns:saturation}
%% ──────────────────────────────────────────────────────────────────

\begin{proposition}[Saturation Threshold]\label{prop:saturation}
For $\theta = 1$:
\begin{enumerate}
\item[\textup{(i)}]  $E_{\mathrm{BS}} \ge 1 \implies  \gamma^* \ge 1/2$
  and $\nu_{\mathrm{eff}} \ge 3\nu_0/2$.
\item[\textup{(ii)}] $E_{\mathrm{BS}} \ge 99 \implies \gamma^* \ge 0.99$
  and $\nu_{\mathrm{eff}} \ge 1.99\,\nu_0$.
\end{enumerate}
\end{proposition}

\begin{proof}
$\gamma^*(E) = E/(E+1)$ is monotone increasing.
$\gamma^*(1)=1/2$; $\gamma^*(99)=99/100$.
\end{proof}

\paragraph{Computational observation (VERIFIED).}
Across all 15 values of Re in the sweep
($314 \le \mathrm{Re} \le 62{,}832$), the maximum observed
$\gamma^*$ is $0.9959 \pm 0.001$.  Saturation occurs within
$O(1)$ non-dimensional time units.  This means the adaptive
system operates at $\nu_{\mathrm{eff}} \approx 2\nu_0$ for
the vast majority of the simulation.

%% ──────────────────────────────────────────────────────────────────
\subsection{Effective Reynolds Number Reduction}
\label{app:ns:re_reduction}
%% ──────────────────────────────────────────────────────────────────

\begin{theorem}[Reynolds Number Halving]\label{thm:re_halving}
Under the adaptive system~\eqref{eq:app_ns}, at saturation
$\gamma^* \to 1$, the effective Reynolds number is
\[
  \mathrm{Re}_{\mathrm{eff}} = \frac{UL}{\nu_{\mathrm{eff}}}
  \approx \frac{\mathrm{Re}_0}{2}.
\]
The adaptive mechanism's primary action is a
factor-$2$ reduction of Reynolds number.
\end{theorem}

\begin{proof}
At saturation, $\nu_{\mathrm{eff}} = \nu_0(1 + \gamma^*) \to 2\nu_0$.
Then $\mathrm{Re}_{\mathrm{eff}} = UL/(2\nu_0) = \mathrm{Re}_0/2$.
\end{proof}

\begin{remark}
This theorem identifies the adaptive viscosity as physically
equivalent to doubling the fluid's kinematic viscosity.  Any
stability result at the adaptive viscosity level is therefore
a stability result at approximately half the nominal Reynolds
number---expected physics, not a novel mathematical mechanism.
\end{remark}

%% ──────────────────────────────────────────────────────────────────
\subsection{The Damping Inequality and Regime Classification}
\label{app:ns:damping}
%% ──────────────────────────────────────────────────────────────────

We consider the empirical inequality
\begin{equation}\label{eq:app_ineq}
  \frac{d}{dt}\norm{\bm{\omega}}_\infty
  \le -c\,\nu\,\norm{\bm{\omega}}_\infty^2 + C^*
\end{equation}
where $c$ and $C^*$ are obtained by least-squares fit of the finite
differences $\Delta\norm{\bm{\omega}}_\infty/\Delta t$ against
$\norm{\bm{\omega}}_\infty^2$, i.e.,
\[
  c = -\frac{\text{slope}}{\nu},\qquad
  C = \text{intercept},\qquad
  C^* = \max_{t_k}\bigl[\dot{\omega}_k + c\,\nu\,\omega_k^2\bigr].
\]

\paragraph{Critical methodological note.}
The upper bound $C^*$ is defined as the \emph{maximum of the
residuals}.  By construction, $C^* = \max_k r_k$ guarantees
zero violations ($r_k \le C^*$ for all $k$).
\textbf{This is a definition, not a verification.} A genuine
verification would require $C^*$ to be determined \emph{a priori}
(e.g., from initial data alone) and then checked against the
trajectory.

\begin{theorem}[Regime Classification by Damping
  Coefficient]\label{thm:regime}
For the Taylor--Green vortex at $N=128$, $T=5.0$, the sign of the
fitted damping coefficient $c(\nu)$ delineates two regimes:
\begin{center}
\begin{tabular}{@{}lccl@{}}
\toprule
Viscosity & $\nu$ & $c$ & Regime \\
\midrule
Constant (bare) & $0.005$ & $-11.91$ & Anti-damping ($c < 0$) \\
Constant (at saturation) & $0.00995$ & $+2.447$ & Damping ($c > 0$) \\
\bottomrule
\end{tabular}
\end{center}
\end{theorem}

\begin{proof}[\textbf{VERIFIED} computationally]
Direct simulation output.  The bare-$\nu$ run
($\nu=0.005$, $\mathrm{Re}\approx 1257$, $N=128$, $T=5.0$,
$\mathrm{dt}=5\times 10^{-4}$, 999~diagnostic snapshots)
yields $c=-11.91$, $R^2=0.652$, $p=8.65\times 10^{-231}$.
The frozen-$\gamma^*$ run ($\nu=0.00995$, identical parameters)
yields $c=+2.447$, $R^2=0.939$, $p<10^{-100}$.
\end{proof}

\begin{theorem}[Critical Viscosity Threshold]\label{thm:nu_crit}
There exists $\nu_c \in (0.005,\, 0.00995)$ such that
\[
  c(\nu_c) = 0.
\]
For $\nu > \nu_c$, the quadratic damping inequality~\eqref{eq:app_ineq}
holds with $c > 0$.  For $\nu < \nu_c$, $c < 0$ and the inequality
provides no bound.
\end{theorem}

\begin{proof}
The damping coefficient $c(\nu)$ depends continuously on $\nu$
(via continuous dependence of solutions on parameters for smooth
NS solutions, plus continuity of the least-squares fit).
Since $c(0.005) = -11.91 < 0$ and $c(0.00995) = +2.447 > 0$,
the intermediate value theorem gives $\nu_c \in (0.005, 0.00995)$
with $c(\nu_c) = 0$.
\end{proof}

\begin{corollary}[Why Adaptive Viscosity Works]\label{cor:why_adaptive}
The adaptive mechanism succeeds because $\gamma^*_{\mathrm{sat}}
\approx 0.996$ pushes $\nu_{\mathrm{eff}} = \nu_0(1+\gamma^*_{\mathrm{sat}})
\approx 0.00995 > \nu_c$, placing the system in the damping regime.
At the bare viscosity $\nu_0 = 0.005 < \nu_c$, the quadratic
damping inequality fails.
\end{corollary}

\begin{remark}[Sign of $c$ Does Not Determine Blow-Up]
\label{rem:c_sign}
$c < 0$ does \emph{not} imply finite-time blow-up.  The
constant-$\nu$ simulation at $\mathrm{Re}\approx 1257$
($c=-11.91$) remains smooth throughout $T=5.0$, with
peak $\norm{\bm{\omega}}_\infty = 10.19$---finite, though
still growing at $t=4.995$.  The sign of $c$ characterises
the utility of the \emph{specific quadratic inequality form}
\eqref{eq:app_ineq}, not the regularity of the solution.
Boundedness may still hold via higher-order nonlinear
saturation mechanisms that the quadratic ansatz cannot capture.
\end{remark}

%% ──────────────────────────────────────────────────────────────────
\subsection{Reynolds Number Sweep: Suppression Scaling Law}
\label{app:ns:sweep}
%% ──────────────────────────────────────────────────────────────────

\paragraph{Setup (VERIFIED).}
15 values of $\nu$ log-spaced from $0.02$ to $10^{-4}$
($\mathrm{Re} \in [314,\, 62{,}832]$), each run in both
constant and adaptive modes at $N=128$, $T=3.0$.

Define the suppression factor
\[
  S(\mathrm{Re}) = \frac{\max_t \mathcal{E}_{\mathrm{const}}(t)}
  {\max_t \mathcal{E}_{\mathrm{adapt}}(t)}.
\]

\begin{theorem}[Suppression Scaling Law]\label{thm:suppression}
In the tested range $314 \le \mathrm{Re} \le 62{,}832$:
\begin{equation}\label{eq:scaling}
  S(\mathrm{Re}) = 1.661\,\mathrm{Re}^{-0.047},\qquad
  R^2 = 0.546,\quad p = 1.64\times 10^{-3}.
\end{equation}
\end{theorem}

\begin{proof}[\textbf{VERIFIED} computationally]
Log-linear regression on 15 data points.  The fit is
statistically significant ($p < 0.01$) but of moderate
explanatory power ($R^2 = 0.546$).
\end{proof}

\begin{corollary}[Suppression Vanishes at High Re]
\label{cor:suppression_vanishes}
The exponent $\alpha = -0.047 < 0$ means suppression
\textbf{decreases} with Reynolds number:
\begin{center}
\begin{tabular}{@{}rrl@{}}
\toprule
Re & $S(\mathrm{Re})$ & Interpretation \\
\midrule
$314$ & $1.045$ & $4.5\%$ suppression \\
$1{,}257$ & $1.025$ & $2.5\%$ suppression \\
$62{,}832$ & $1.008$ & $0.8\%$ suppression \\
$10^6$ & $\approx 1.002$ & $0.2\%$ (extrapolated) \\
\bottomrule
\end{tabular}
\end{center}
At turbulent Reynolds numbers ($\mathrm{Re} \sim 10^6$),
the factor-$2$ viscosity enhancement from the adaptive mechanism
produces negligible enstrophy suppression.
\end{corollary}

\begin{remark}
This does \emph{not} mean the adaptive mechanism ``fails'' at
high Re.  It means the peak-enstrophy \emph{ratio} between
constant and adaptive modes converges to~$1$---both modes
produce nearly identical peak enstrophy.  In absolute terms,
the adaptive system may still provide useful diagnostic
information.  However, the margin for stability improvement
shrinks.
\end{remark}

\paragraph{Key subsidiary observations (VERIFIED).}
\begin{enumerate}
\item \textbf{No blow-up in either mode.}  All 30 simulations
  (15~constant + 15~adaptive) produce smooth solutions through
  $T=3.0$.  Peak enstrophy is finite at every Re.

\item \textbf{BKM integral finite.}  $\int_0^T
  \norm{\bm{\omega}}_\infty\,dt$ is finite and bounded across
  all runs.  BKM reduction from adaptive mode ranges from
  $+13.4\%$ (Re$=314$) to $+0.21\%$ (Re$=62{,}832$),
  monotonically decreasing with Re.

\item \textbf{$\gamma^*$ saturates universally.}
  $\gamma^*_{\max} = 0.9959$ with negligible variation
  across Re.
\end{enumerate}

%% ──────────────────────────────────────────────────────────────────
\subsection{Constant-Viscosity Run: Detailed Analysis}
\label{app:ns:constant}
%% ──────────────────────────────────────────────────────────────────

\paragraph{Setup.}  $N=128$, $\nu_0=0.005$ ($\mathrm{Re}\approx 1257$),
$\mathrm{dt}=5\times 10^{-4}$, $T=5.0$, $\gamma^* \equiv 0$
(no adaptation), diagnostic interval $= 10$ steps
($999$ snapshots).

\paragraph{Results (VERIFIED).}
\begin{enumerate}
\item $c = -11.91$ (negative---\emph{anti-damping}).
\item $C^* = 2.272$, $R^2 = 0.652$, $p = 8.65\times 10^{-231}$.
\item Peak $\norm{\bm{\omega}}_\infty = 10.19$ at $t = 4.995$
  (the final snapshot---vorticity is \emph{still increasing}
  when the simulation ends).
\item Kinetic energy: $\mathrm{KE}_0 = 0.125 \to \mathrm{KE}_f$
  (decaying, as expected).
\item BKM integral finite but growing.
\end{enumerate}

\begin{remark}[Implications for Regularity Claims]
\label{rem:constant_implications}
Any claim that ``the differential inequality holds at constant
$\nu$'' must contend with this data:
\begin{enumerate}
\item[\textup{(a)}] $c < 0$ at $\nu=0.005$ means the quadratic
  inequality~\eqref{eq:app_ineq} is \emph{reversed}.  The $\omega^2$
  term \emph{amplifies} rather than damps growth.

\item[\textup{(b)}] The prior claim of $c = +2.447$ at ``constant
  $\nu$'' used $\nu = 0.00995 = \nu_0(1 + \gamma^*_{\mathrm{sat}})$,
  which is the frozen adaptive saturation level---effectively
  \emph{twice} the bare viscosity.  This corresponds to
  $\mathrm{Re}_{\mathrm{eff}} \approx 631$, not $\mathrm{Re} = 1257$.

\item[\textup{(c)}] The solution at $\nu=0.005$ does \emph{not}
  blow up in $T=5.0$, but vorticity is still growing at
  the final snapshot.  Extending $T$ would determine whether
  the growth saturates or continues.
\end{enumerate}
\end{remark}

\subsection{BSDT Channel Behaviour at Constant Viscosity}
\label{app:ns:channels_const}
%% ──────────────────────────────────────────────────────────────────

\paragraph{Observations (VERIFIED).}
At $\nu = 0.005$ (constant), $T=5.0$:
\begin{enumerate}
\item $\delta_G$: largest at $t \approx 0.005$ (initialisation
  transient---the Taylor--Green spectrum is far from $k^{-5/3}$),
  then decreases as the cascade develops.  The ``early spike''
  is an artefact of the reference spectrum choice, not a
  predictive signal.

\item $\delta_C$: z-score of enstrophy.  Grows monotonically
  as enstrophy departs from its running mean.  This is a
  \emph{tautological reporter}: it detects enstrophy growth
  because it \emph{is} enstrophy growth (normalised).

\item $\delta_T$: spikes at $t = 0.005$ (first time step),
  then decreases.  Again an initialisation transient.

\item $\delta_A \equiv 0$ (not implemented in the solver).
\end{enumerate}

The observed ``early spikes'' in $\delta_G$ and $\delta_T$
are initialisation artefacts at $t = \Delta t$, not evidence
of predictive early-warning capability.

\subsection{The Gap: From Adaptive to Classical Regularity}
\label{app:ns:gap}
%% ──────────────────────────────────────────────────────────────────

The central question is whether regularity of the adaptive
system~\eqref{eq:app_ns} implies regularity of the classical
system (constant $\nu_0$).  The simulation data delineates the gap
precisely.

\begin{theorem}[Three-Regime Structure]\label{thm:three_regimes}
The simulations identify three distinct regimes:
\begin{center}
\begin{tabular}{@{}lccccl@{}}
\toprule
System & $\nu$ & $\mathrm{Re}$ & $c$ & Inequality & Status \\
\midrule
Adaptive & varies in $[\nu_0, 2\nu_0)$ & $\approx 631$--$1257$
  & $> 0$ & Holds & VERIFIED \\
Frozen $\gamma^*$ (constant) & $0.00995$ & $\approx 631$
  & $+2.447$ & Holds & VERIFIED \\
Classical (constant) & $0.005$ & $\approx 1257$
  & $-11.91$ & FAILS & VERIFIED \\
\bottomrule
\end{tabular}
\end{center}
\end{theorem}

\begin{proof}
Direct simulation output as reported above.
\end{proof}

\paragraph{The gap in one sentence.}  The adaptive mechanism works
by pushing $\nu_{\mathrm{eff}}$ above the critical threshold
$\nu_c$.  Any ``bridge'' from adaptive to classical regularity
must explain how to cross from $\mathrm{Re}\approx 631$ (where
$c > 0$) to $\mathrm{Re}\approx 1257$ (where $c < 0$)---i.e.,
it must provide \emph{a priori} bounds that the quadratic
inequality~\eqref{eq:app_ineq} cannot.

%% ──────────────────────────────────────────────────────────────────
\subsection{Conjectures and Open Problems}
\label{app:ns:conjectures}
%% ──────────────────────────────────────────────────────────────────

The simulation data motivates several open problems.

\begin{conjecture}[Critical Viscosity Function]\label{conj:nu_c}
For a given initial condition and resolution, define
$\nu_c(\mathrm{Re}, \bm{u}_0)$ as the viscosity at which the
fitted damping coefficient $c(\nu) = 0$.  Then as the resolution
$N \to \infty$ with fixed IC:
\begin{enumerate}
\item[\textup{(i)}]
  $\nu_c$ converges to a resolution-independent limit
  (well-posedness of the critical threshold).
\item[\textup{(ii)}]
  The ratio $\nu_c / \nu_0$ determines whether the factor-$2$
  adaptive enhancement suffices for all Re.  If
  $\nu_c / \nu_0 < 2$ universally, then $\nu_{\mathrm{eff}}
  > \nu_c$ at saturation and the damping inequality holds
  under adaptation at every Re.
\item[\textup{(iii)}]
  If $\nu_c / \nu_0 \to \infty$ as $\mathrm{Re} \to \infty$,
  then the adaptive mechanism's fixed factor-$2$ enhancement
  becomes insufficient at high Re.
\end{enumerate}
\end{conjecture}

\paragraph{Testability.}  This conjecture can be resolved
computationally by running intermediate viscosities (e.g.,
$\nu \in \{0.006, 0.007, 0.008, 0.009\}$) to locate $\nu_c$,
then repeating at multiple Re values to determine the scaling.

\begin{conjecture}[Vorticity Saturation at Constant
  Viscosity]\label{conj:saturation}
At $\nu = 0.005$ ($\mathrm{Re}\approx 1257$), the vorticity
$\norm{\bm{\omega}}_\infty(t)$ reaches a finite maximum at some
$t^* > 5.0$ and subsequently decays.  I.e., the still-growing
vorticity at $t=5.0$ is a transient, not incipient blow-up.
\end{conjecture}

\paragraph{Testability.}  Extend the simulation to $T=20$ or
$T=50$.  If $\norm{\bm{\omega}}_\infty$ peaks and decays,
the conjecture is confirmed.  If growth continues or accelerates,
the conjecture is refuted.

\begin{conjecture}[P/D Halving and Adaptive Feedback]
\label{conj:pd}
The production-to-dissipation ratio
$P/D = \int\omega_i S_{ij}\omega_j\,dx /
(\nu_{\mathrm{eff}}\norm{\grad\bm{\omega}}_{L^2}^2)$
converges to $1/2$ under adaptive viscosity (as reported in
Section~\ref{sec:ns:lyapv2}).  Under constant viscosity at
$\nu < \nu_c$, the ratio does \textbf{not} converge to $1/2$.

If true, the P/D halving is a property of the feedback control,
not of the Navier--Stokes equations per se. If false (i.e., if
$P/D \to 1/2$ at constant $\nu$ as well), then the halving is an
intrinsic property of 3D NS turbulence and adaptive viscosity
merely accelerates convergence.
\end{conjecture}

%% ──────────────────────────────────────────────────────────────────
\subsection{Honest Assessment: What These Results Establish}
\label{app:ns:assessment}
%% ──────────────────────────────────────────────────────────────────

We summarise what is proved, what is computationally verified,
and what remains open.

\paragraph{Proved (from definitions and standard analysis).}
\begin{enumerate}
\item The adaptive viscosity satisfies $\nu_0 \le \nu_{\mathrm{eff}}
  < 2\nu_0$ (Proposition~\ref{prop:nu_bounds}).
\item The adaptive system has Leray--Hopf weak solutions.
\item At saturation, $\mathrm{Re}_{\mathrm{eff}} \approx
  \mathrm{Re}_0 / 2$ (Theorem~\ref{thm:re_halving}).
\item There exists $\nu_c \in (0.005, 0.00995)$ where the
  damping coefficient changes sign
  (Theorem~\ref{thm:nu_crit}).
\end{enumerate}

\paragraph{Computationally verified (reproducible, subject to
  numerical resolution).}
\begin{enumerate}
\item $\gamma^*$ saturates at $\approx 0.996$ across all tested Re
  (Proposition~\ref{prop:saturation}).
\item The damping inequality~\eqref{eq:app_ineq} holds with
  $c > 0$ at $\nu = 0.00995$ ($\mathrm{Re}_{\mathrm{eff}}
  \approx 631$) and fails ($c < 0$) at $\nu = 0.005$
  ($\mathrm{Re} \approx 1257$) (Theorem~\ref{thm:regime}).
\item Suppression decreases with Re: $S \sim
  \mathrm{Re}^{-0.047}$ (Theorem~\ref{thm:suppression}).
\item No blow-up in any of 30 simulations through $T = 3$--$5$,
  $\mathrm{Re}$ up to $62{,}832$.
\item The BKM integral remains finite at all tested parameters.
\end{enumerate}

\paragraph{Not established (open).}
\begin{enumerate}
\item Global regularity for classical constant-$\nu$ NS
  (the data show $c < 0$ at the paper's Re).
\item Global regularity for adaptive NS with arbitrary initial
  data (only Taylor--Green tested).
\item Whether $\norm{\bm{\omega}}_\infty$ at $\nu=0.005$
  eventually peaks or continues growing past $T=5$.
\item Whether $\nu_c / \nu_0 < 2$ at all Re
  (Conjecture~\ref{conj:nu_c}).
\item Any bridge from adaptive to classical regularity.
\end{enumerate}

\paragraph{Previously overclaimed (corrected here).}
\begin{enumerate}
\item The ``$c = 2.447$ at constant $\nu$'' result used
  $\nu = 0.00995 \approx 2\nu_0$, not the bare $\nu_0 = 0.005$.
  This is a result at $\mathrm{Re}\approx 631$, not
  $\mathrm{Re}\approx 1257$.

\item ``0 violations'' of the inequality is tautological
  when $C^*$ is defined as the maximum residual.

\item ``Early spikes'' in $\delta_G$ and $\delta_T$ at constant
  $\nu$ are initialisation transients, not predictive early
  warnings.
\end{enumerate}

%% ──────────────────────────────────────────────────────────────────
\subsection{The Road Forward}
\label{app:ns:road}
%% ──────────────────────────────────────────────────────────────────

The critical viscosity threshold $\nu_c$
(Theorem~\ref{thm:nu_crit}) is the key object for future
work.  Three computational experiments would clarify the
path to stronger results:

\begin{enumerate}
\item \textbf{Locate $\nu_c$ precisely.}  Run 10 intermediate
  viscosities $\nu \in (0.005, 0.00995)$ at $N=128$ to find
  where $c(\nu)$ crosses zero.  A refined estimate
  $\nu_c \approx \nu_0(1 + r)$ determines the required
  viscosity enhancement ratio $r$.

\item \textbf{Extend $T$ at $\nu = 0.005$.}  Run to $T = 20$
  or $T = 50$ to determine whether peak vorticity saturates
  (Conjecture~\ref{conj:saturation}).  If it does, the
  quadratic inequality is the wrong diagnostic---the system
  is stable for reasons the inequality cannot capture.

\item \textbf{Scale $\nu_c$ with Re.}  Repeat experiment~1 at
  Re $= 3{,}000$, $10{,}000$, $30{,}000$ to determine whether
